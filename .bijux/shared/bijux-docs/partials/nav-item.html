{%- macro render(nav_item, path, level, authored=none, page_occurrences=none) -%} {%- set label = nav_item.title -%} {%- set authored_children = none -%} {%- if authored is mapping and authored | length == 1 -%} {%- set authored_name = authored | first -%} {%- set authored_value = authored[authored_name] -%} {%- if nav_item.children and authored_name == nav_item.title -%} {%- if authored_value is mapping or (authored_value is sequence and authored_value is not string) -%} {%- set authored_children = authored_value -%} {%- endif -%} {%- elif authored_value is string -%} {# Page objects can be reused by MkDocs; an occurrence's label belongs to its matched authored source, not the first shared Page title. #} {%- if nav_item.is_page and authored_value.lstrip("/") == nav_item.file.src_uri -%} {%- if page_occurrences and page_occurrences.get(nav_item.file.src_uri, 0) > 1 -%} {%- set label = authored_name | e -%} {%- endif -%} {%- endif -%} {%- endif -%} {%- endif -%}
  • {%- if nav_item.children -%}
    {{ label }}
      {%- for child in nav_item.children -%} {%- set child_authored = none -%} {%- if authored_children is mapping -%} {%- set names = authored_children | list -%} {%- if loop.index0 < names | length -%} {%- set name = names[loop.index0] -%} {%- set child_authored = {name: authored_children[name]} -%} {%- endif -%} {%- elif authored_children is sequence and loop.index0 < authored_children | length -%} {%- set child_authored = authored_children[loop.index0] -%} {%- endif -%} {{ render(child, path ~ "-" ~ loop.index, level + 1, child_authored, page_occurrences) }} {%- endfor -%}
    {%- else -%} {{ label }} {%- endif -%}
